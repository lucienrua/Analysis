% 核心宏：一键绑定 thmtools 样式与 tcolorbox 外观
% #1: 环境命令名 (如 theorem)
% #2: 环境显示名 (如 Theorem)
% #3: 颜色变量名 (如 ObsTheorem)
% #4: 计数器规则 (如 numberwithin=section 或 sibling=theorem)
\newcommand{\createobsbox}[4]{
    % 1. 配置专属 thmtools 样式 (负责标题颜色、括号格式与换行)
    \declaretheoremstyle[
        headfont=\bfseries\color{#3},      % 主标题同色加粗
        notefont=\bfseries\color{#3},      % 括号内的小标题同色加粗
        notebraces={(}{)},                 % 小标题使用括号包裹 (匹配你图片中的效果)
        bodyfont=\normalfont,
        spaceabove=0pt, spacebelow=0pt,
        headpunct={.},                     % 标题末尾的点号
        postheadspace=\newline             % 强制在标题结束后换行
    ]{#1style}

    % 2. 声明定理环境
    \declaretheorem[style=#1style, name=#2, #4]{#1}
    \declaretheorem[style=#1style, name=#2, numbered=no]{#1*}

    % 3. 绑定 tcolorbox 样式 (只负责外框和背景)
    \tcolorboxenvironment{#1}{
        enhanced, breakable, rounded corners, boxrule=0.5pt,
        colframe=#3,           % 边框使用对应颜色
        colback=#3!10,         % 背景使用 10% 透明度的对应颜色
        left=1em, right=1em, top=0.5em, bottom=0.5em,
        before skip=10pt, after skip=10pt,
        separator sign none
    }
    \tcolorboxenvironment{#1*}{
        enhanced, breakable, rounded corners, boxrule=0.5pt,
        colframe=#3, colback=#3!10,
        left=1em, right=1em, top=0.5em, bottom=0.5em,
        before skip=10pt, after skip=10pt,
        separator sign none
    }
}
